# Frame-level yes/no checks.
#
#   [name] severity=high threshold=0.70 requires=motorcycle tiles=no
#     + positive phrase
#     - negative phrase
#
#   requires=        check does not run unless one of these objects was found
#   requires_people= minimum person count
#   tiles=no         whole frame only (for global properties like darkness —
#                    one dark tile in a bright image is a dark jacket)
#
# Gating is what stops "no helmet 99%" firing on a studio portrait. If a
# check can only be true when some object is present, say so here.

# ============================ FIRE / SMOKE ============================

[fire] severity=high threshold=0.72
+ a photograph of a fire burning
+ a photograph of orange flames
+ a building on fire
+ a vehicle on fire
- a normal scene with no fire
- an ordinary photograph with no flames
- a scene with orange or yellow objects but no fire
- a sunset sky with no fire
- a person wearing orange clothing

[smoke] severity=high threshold=0.72
+ a photograph filled with thick smoke
+ smoke rising from a fire
+ a room filled with grey smoke
- a normal clear scene with no smoke
- a hazy or foggy scene with no smoke
- an ordinary photograph with clear air
- a plain grey background

# ============================ THEFT / SAFETY ===========================
# Purpose-built for the residential-society and street context: snatching,
# vehicle break-ins, and an unsupervised child near a road or gate.

+ a person snatching a bag, phone or chain from another person
+ a robbery or mugging in progress on the street
- people greeting or shaking hands
