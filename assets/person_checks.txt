# Per-person checks — run on each PERSON CROP, not the whole frame.
#
# This is how you catch a weapon without ever detecting the weapon. A machete
# held by a man is inside the man's box, so the person crop already contains
# it. No proposal box for the blade is required.
#
# Same contrastive format as anomalies.txt: positives vs negatives, softmax
# between the averaged sides gives P(true).

[weapon_held] severity=high threshold=0.60
+ a person holding a large knife
+ a person holding a machete
+ a person holding a sword
+ a person holding a gun
+ a person holding an iron rod as a weapon
+ a person holding a wooden stick as a weapon
- a person with empty hands
- a person holding a mobile phone
- a person holding a bag
- a person holding an umbrella
- a person holding a bottle
- a person holding a work tool
- a person holding papers

- a person standing calmly
- a person walking normally
- a person talking to someone

[on_ground] severity=high threshold=0.65

- a person sitting on a chair
